export function calendarValue(date: Date, precision: CalendarPrecision): string {
  const month = `${String(date.getFullYear()).padStart(4, '0')}-${pad(date.getMonth() + 1)}`;
  return precision === 'month' ? month : `${month}-${pad(date.getDate())}`;
}
export function validCalendarValue(value: string, precision: CalendarPrecision): boolean {
  return validateCalendarTimeRange({ from: value, to: value }, precision).length === 0;
}
export function calendarView(value: string, today = new Date()): CalendarView {
  return { year: Number(value.slice(0, 4)) || today.getFullYear(), month: Number(value.slice(5, 7)) || today.getMonth() + 1 };
}
export function shiftCalendar(view: CalendarView, months: number): CalendarView {
  const index = view.year * 12 + view.month - 1 + months;
  return { year: Math.floor(index / 12), month: ((index % 12) + 12) % 12 + 1 };
}
function localDate(year: number, month: number, day: number): Date {
  const date = new Date(0);
  date.setFullYear(year, month - 1, day);
  date.setHours(12, 0, 0, 0);
  return date;
}
export function calendarDays(view: CalendarView): CalendarCell[][] {
  const weekday = localDate(view.year, view.month, 1).getDay();
  // Tiny includes the previous week's Sunday when the month begins on Sunday.
  const offset = weekday || 7;
  return Array.from({ length: 6 }, (_, row) => Array.from({ length: 7 }, (_, col) => {
    const date = localDate(view.year, view.month, row * 7 + col + 1 - offset);
    const value = calendarValue(date, 'date');
    const viewMonth = `${view.year}-${pad(view.month)}`;
    return { value, label: date.getDate(), adjacent: value.slice(0, 7) < viewMonth ? 'pre-month' : value.slice(0, 7) > viewMonth ? 'next-month' : null };
  }));
}
export function calendarMonths(year: number): CalendarCell[][] {
  return Array.from({ length: 3 }, (_, row) => Array.from({ length: 4 }, (_, col) => {
    const month = row * 4 + col + 1;
    return { value: `${String(year).padStart(4, '0')}-${pad(month)}`, label: month, adjacent: null };
  }));
}
export function moveCalendarValue(value: string, precision: CalendarPrecision, amount: number): string {
  const view = calendarView(value);
  return calendarValue(localDate(view.year, view.month + (precision === 'month' ? amount : 0), precision === 'date' ? Number(value.slice(8)) + amount : 1), precision);
}
export function calendarDraft(from: string, to: string, precision: CalendarPrecision, range: boolean, required: boolean):
  { ok: true; value: { from: string; to: string } | null } | { ok: false; error: string } {
  if (!from && (!range || !to)) return required ? { ok: false, error: '请选择日期' } : { ok: true, value: null };
  if (range && (!from || !to)) return { ok: false, error: '请填写完整的起止期间' };
  const value = { from, to: range ? to : from };
  const error = validateCalendarTimeRange(value, precision)[0]?.message;
  return error ? { ok: false, error } : { ok: true, value };
}

import { calendarDays, calendarDraft, calendarMonths, calendarValue, moveCalendarValue, shiftCalendar, validCalendarValue } from '../src/filters/date-picker';

describe('日期选择器业务日期与提交边界', () => {
  it('闰年及月底严格校验，不把非法日期归一化', () => {
    expect(validCalendarValue('2024-02-29', 'date')).toBe(true);
    for (const value of ['2026-02-29', '2026-04-31', '2026-13-01', '2026-1-01', 'bad']) {
      expect(validCalendarValue(value, 'date')).toBe(false);
    }
    expect(validCalendarValue('2026-12', 'month')).toBe(true);
    expect(validCalendarValue('2026-13', 'month')).toBe(false);
  });
  it('半范围及倒置范围无可应用值', () => {
    expect(calendarDraft('2026-01', '', 'month', true, false).ok).toBe(false);
    expect(calendarDraft('', '2026-06', 'month', true, false).ok).toBe(false);
    expect(calendarDraft('2026-06', '2026-01', 'month', true, false).ok).toBe(false);
    expect(calendarDraft('2026-02-29', '2026-03-01', 'date', true, false).ok).toBe(false);
  });
  it('同日及跨年完整范围提交原字符串', () => {
    expect(calendarDraft('2026-12', '2027-01', 'month', true, false)).toEqual({ ok: true, value: { from: '2026-12', to: '2027-01' } });
    expect(calendarDraft('2024-02-29', '2024-02-29', 'date', true, true)).toEqual({ ok: true, value: { from: '2024-02-29', to: '2024-02-29' } });
  });
  it('单值不依赖范围终点；required 不允许清空', () => {
    expect(calendarDraft('2026-01', '', 'month', false, true)).toEqual({ ok: true, value: { from: '2026-01', to: '2026-01' } });
    expect(calendarDraft('', '', 'date', false, true).ok).toBe(false);
    expect(calendarDraft('', '', 'date', true, true).ok).toBe(false);
    expect(calendarDraft('', '', 'date', true, false)).toEqual({ ok: true, value: null });
  });
  it('日历显示周日开头、相邻月份与闰日，42 个连续业务日期', () => {
    const rows = calendarDays({ year: 2024, month: 2 });
    expect(rows).toHaveLength(6);
    expect(rows.every(row => row.length === 7)).toBe(true);
    expect(rows[0][0]).toEqual({ value: '2024-01-28', label: 28, adjacent: 'pre-month' });
    expect(rows.flat().find(cell => cell.value === '2024-02-29')?.adjacent).toBe(null);
    expect(rows[5][6].value).toBe('2024-03-09');
    expect(new Set(rows.flat().map(cell => cell.value)).size).toBe(42);
  });
  it('翻页与键盘跨年/闰日不经过 UTC 序列化', () => {

    expect(moveCalendarValue('2024-02-29', 'date', 1)).toBe('2024-03-01');
    expect(calendarMonths(2027).flat().map(cell => cell.value)).toEqual(Array.from({ length: 12 }, (_, i) => `2027-${String(i + 1).padStart(2, '0')}`));
    expect(calendarValue(new Date(2026, 0, 31, 23, 59), 'date')).toBe('2026-01-31');
  });
